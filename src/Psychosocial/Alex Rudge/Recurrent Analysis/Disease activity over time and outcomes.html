Introduction

This analysis addresses whether change in disease activity over time is associated with subsequent flare outcomes, using IBD-Control-VAS (OverallControl) as the measure of disease activity. Unlike control_8 (the 8-item IBD-Control sub-scale), which is only available as a single baseline value per participant, OverallControl is collected both at baseline (Baseline2022/IBD.xlsx) and monthly during follow-up (monthlyQ.xlsx), making it the only IBD-Control measure that can be modelled as a genuinely time-varying covariate.

OverallControl is added as a time-dependent confounder to each of the main analysis’s extended Cox models (Longitudinal analysis/{HADS,PHQ,PSQI, Exercise,Fatigue,Life Events}/), rather than modelled as a standalone exposure. Every risk factor already covered by the main analysis is re-fitted here with time-dependent OverallControl added alongside its own time-dependent exposure, for Crohn’s Disease and Ulcerative Colitis/IBDU separately, for both patient-reported and objective flare - the same structure and covariate set as the reference models (IMD + Sex + age_decade + FC + Smoke + frailty(SiteNo)), with OverallControl_per10 added to each formula. OverallControl enters per 10-point change (of the 0-100 scale) for an interpretable hazard ratio.

# Takes already-tidied, already-filtered data (not the raw model) -
# run_cox_cell() below tidies each fit once and filters to the risk
# factor's own term(s), so confounders (including OverallControl, which is
# fitted for adjustment only) never reach this table.
print_survival <- function(tidy_data) {
  tidy_data %>%
    dplyr::select(term, estimate, std.error, statistic, conf.low, conf.high, p.value) %>%
    dplyr::mutate(dplyr::across(.cols = where(is.numeric), .fns = ~ signif(.x, 3))) %>%
    dplyr::mutate(p.value = as.character(p.value)) %>%
    knitr::kable(
      col.names = c("Covariate", "Hazard Ratio", "Std. Error", "Statistic", "Lower 95% CI", "Upper 95% CI", "P-value"),
      format.args = list(drop0trailing = TRUE)
    )
}

Base survival endpoint

flares_soft.RDS/flares_hard.RDS provide the first-flare survival endpoint (time, DiseaseFlareYN), the same source used for the main analysis’s Cox models. Built before sourcing any risk factor’s data_cleaning.R below, since some of those scripts also define flares_soft/flares_hard as a side effect - harmless (identical source files, reloaded), but this keeps the base data unambiguous.

  dplyr::inner_join(flares_hard %>% dplyr::select(ParticipantNo, hardflare, hardflare_time), by = "ParticipantNo") %>%
  dplyr::transmute(
    ParticipantNo, SiteNo, diagnosis2, Sex, age_decade, FC, Smoke, IMD,

  dplyr::arrange(ParticipantNo, month) %>%
  dplyr::group_by(ParticipantNo) %>%
  tidyr::fill(OverallControl, .direction = "down") %>%
  dplyr::ungroup() %>%
  dplyr::transmute(ParticipantNo, day = month * 365 / 12, OverallControl_per10 = OverallControl / 10)

Model-fitting and reporting helpers

Each risk factor’s own time-dependent score keeps a generic score_group name (matching the recurrent-event analogue in Poisson Analysis - Time-varying covariates.qmd), so one set of helpers covers every exposure below.

Code 
# (ParticipantNo, month, score_group) -> the day-based, LOCF-filled format
# tdc() needs.
exposure_to_day <- function(exposure_long) {
  exposure_long %>%
    dplyr::distinct(ParticipantNo, month, .keep_all = TRUE) %>%
    dplyr::arrange(ParticipantNo, month) %>%
    dplyr::group_by(ParticipantNo) %>%
    tidyr::fill(score_group, .direction = "down") %>%
    dplyr::ungroup() %>%
    dplyr::transmute(ParticipantNo, day = as.numeric(month) * 365 / 12, score_group)
}

# Start-stop Cox dataset for one flare type, adding both the risk factor's
# own time-dependent score and OverallControl_per10 as a second
# time-dependent confounder.
build_cox_td <- function(cox_base, exposure_day) {
  tmerge(cox_base, cox_base, id = ParticipantNo, endpoint = event(time, DiseaseFlareYN)) %>%
    tmerge(data2 = exposure_day, id = ParticipantNo, score_group = tdc(day, score_group)) %>%
    tmerge(data2 = overall_control_td, id = ParticipantNo, OverallControl_per10 = tdc(day, OverallControl_per10))
}

cox_hazard_ratios <- tibble::tibble()

run_cox_cell <- function(data_merged, dx, exposure_name, outcome_label, diagnosis_label) {
  d <- data_merged %>% dplyr::filter(diagnosis2 == dx)
  fit <- coxph(
    Surv(tstart, tstop, endpoint) ~
      score_group + OverallControl_per10 + IMD + Sex + age_decade + FC + Smoke + frailty(SiteNo),
    data = d
  )
  cat("n =", dplyr::n_distinct(d$ParticipantNo), "\n\n")

  # OverallControl_per10 and the other covariates are fitted for adjustment
  # only - only the risk factor's own term(s) are reported or plotted.
  tidy_exposure_only <- fit %>%
    broom::tidy(exponentiate = TRUE, conf.int = TRUE) %>%
    dplyr::filter(stringr::str_starts(term, "score_group"))

  cox_hazard_ratios <<- dplyr::bind_rows(
    cox_hazard_ratios,
    tidy_exposure_only %>% dplyr::mutate(exposure = exposure_name, outcome = outcome_label, diagnosis = diagnosis_label)
  )

  tidy_exposure_only %>% print_survival()
}

score_hads <- function(df) {
  df %>%
    dplyr::mutate(dplyr::across(tidyselect::all_of(hads_items), ~ . - 1)) %>%
    dplyr::mutate(dplyr::across(tidyselect::all_of(inverse_coded_items), ~ 3 - .)) %>%

      anxiety_hads = FeelTense + FrightenedFeelingSomethingAwful + WorryingThoughts +
        SitAtEase + FrightenedFeelingButterflies + FeelRestless + SuddenFeelingsPanic,
      depression_hads = EnjoyThings + CanLaugh + FeelCheerful + FeelSlowedDown +
        LostInterestAppearance + LookForward + CanEnjoyBookTV
    )
}

Combined forest plot

Each risk factor’s own hazard ratio, adjusted for the main analysis’s usual confounders (IMD, Sex, age_decade, FC, Smoke, frailty(SiteNo)) plus time-dependent OverallControl. cox_hazard_ratios only ever contains each risk factor’s own term(s) - confounders, including OverallControl_per10, are fitted for adjustment only and never enter this table (see run_cox_cell() in Setup).

Code 
cox_hazard_ratios %>%

## Introduction

This analysis addresses whether change in disease activity over time is
associated with subsequent flare outcomes, using IBD-Control-VAS
(`OverallControl`) as the measure of disease activity. Unlike `control_8`
(the 8-item IBD-Control sub-scale), which is only available as a single
baseline value per participant, `OverallControl` is collected both at
baseline (`Baseline2022/IBD.xlsx`) and monthly during follow-up
(`monthlyQ.xlsx`), making it the only IBD-Control measure that can be
modelled as a genuinely time-varying covariate.

`OverallControl` is added as a **time-dependent confounder** to each of the
main analysis's extended Cox models (`Longitudinal analysis/{HADS,PHQ,PSQI,
Exercise,Fatigue,Life Events}/`), rather than modelled as a standalone
exposure. Every risk factor already covered by the main analysis is
re-fitted here with time-dependent `OverallControl` added alongside its own
time-dependent exposure, for Crohn's Disease and Ulcerative Colitis/IBDU
separately, for both patient-reported and objective flare - the same
structure and covariate set as the reference models (`IMD + Sex +
age_decade + FC + Smoke + frailty(SiteNo)`), with `OverallControl_per10`
added to each formula. `OverallControl` enters per 10-point change (of the
0-100 scale) for an interpretable hazard ratio.

## Model-fitting and reporting helpers

Each risk factor's own time-dependent score keeps a generic `score_group`
name (matching the recurrent-event analogue in `Poisson Analysis -
Time-varying covariates.qmd`), so one set of helpers covers every exposure
below.

```{r}
# (ParticipantNo, month, score_group) -> the day-based, LOCF-filled format
# tdc() needs.
exposure_to_day <- function(exposure_long) {
  exposure_long %>%
    dplyr::distinct(ParticipantNo, month, .keep_all = TRUE) %>%
    dplyr::arrange(ParticipantNo, month) %>%
    dplyr::group_by(ParticipantNo) %>%
    tidyr::fill(score_group, .direction = "down") %>%
    dplyr::ungroup() %>%
    dplyr::transmute(ParticipantNo, day = as.numeric(month) * 365 / 12, score_group)
}

# Start-stop Cox dataset for one flare type, adding both the risk factor's
# own time-dependent score and OverallControl_per10 as a second
# time-dependent confounder.
build_cox_td <- function(cox_base, exposure_day) {
  tmerge(cox_base, cox_base, id = ParticipantNo, endpoint = event(time, DiseaseFlareYN)) %>%
    tmerge(data2 = exposure_day, id = ParticipantNo, score_group = tdc(day, score_group)) %>%
    tmerge(data2 = overall_control_td, id = ParticipantNo, OverallControl_per10 = tdc(day, OverallControl_per10))
}

cox_hazard_ratios <- tibble::tibble()

run_cox_cell <- function(data_merged, dx, exposure_name, outcome_label, diagnosis_label) {
  d <- data_merged %>% dplyr::filter(diagnosis2 == dx)
  fit <- coxph(
    Surv(tstart, tstop, endpoint) ~
      score_group + OverallControl_per10 + IMD + Sex + age_decade + FC + Smoke + frailty(SiteNo),
    data = d
  )
  cat("n =", dplyr::n_distinct(d$ParticipantNo), "\n\n")

  # OverallControl_per10 and the other covariates are fitted for adjustment
  # only - only the risk factor's own term(s) are reported or plotted.
  tidy_exposure_only <- fit %>%
    broom::tidy(exponentiate = TRUE, conf.int = TRUE) %>%
    dplyr::filter(stringr::str_starts(term, "score_group"))

  cox_hazard_ratios <<- dplyr::bind_rows(
    cox_hazard_ratios,
    tidy_exposure_only %>% dplyr::mutate(exposure = exposure_name, outcome = outcome_label, diagnosis = diagnosis_label)
  )

  tidy_exposure_only %>% print_survival()
}
```
